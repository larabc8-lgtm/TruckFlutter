descanso_semanal_widget.dart 
1    import 'package:flutter/material.dart';
2    import '../services/api_service.dart';
3    
4    // Widget que muestra el estado del descanso semanal
5    class DescansoSemanalWidget extends StatefulWidget {
6      final int idUsuario;
7      final int? idUltimoRegistro;
8      const DescansoSemanalWidget({
9        required this.idUsuario,
10       this.idUltimoRegistro,
11       super.key,
12     });
13   
14     @override
15     State<DescansoSemanalWidget> createState() => _DescansoSemanalWidgetState();
16   }
17   
18   class _DescansoSemanalWidgetState extends State<DescansoSemanalWidget> {
19     final ApiService _api = ApiService();
20   
21     bool _cargando        = true;
22     int _diasConsecutivos = 0;
23     int _diasHastaLimite  = 6;
24     String _nivelAlerta   = 'ok';
25     List<dynamic> _compensaciones = [];
26     int _totalHorasDeuda  = 0;
27   
28     @override
29     void initState() {
30       super.initState();
31       _cargarEstado();
32     }
33   
34     void _cargarEstado() async {
35       setState(() => _cargando = true);
36       var res = await _api.getEstadoDescansoSemanal(widget.idUsuario);
37       if (!mounted) return;
38       if (res['status'] == 'success') {
39         final data = res['data'];
40         setState(() {
41           _diasConsecutivos = data['dias_consecutivos'] ?? 0;
42           _diasHastaLimite  = data['dias_hasta_limite']  ?? 6;
43           _nivelAlerta      = data['nivel_alerta']        ?? 'ok';
44           _compensaciones   = data['compensaciones_pendientes'] ?? [];
45           _totalHorasDeuda  = data['total_horas_deuda']   ?? 0;
46           _cargando = false;
47         });
48       } else {
49         setState(() => _cargando = false);
50       }
51     }
52   
53     void _confirmarDescansoSemanal() async {
54       if (widget.idUltimoRegistro == null) {
55         ScaffoldMessenger.of(context).showSnackBar(
56           const SnackBar(
57             content: Text("No hay ningún registro de descanso activo para marcar."),
58             backgroundColor: Colors.orange,
59           ),
60         );
61         return;
62       }
63   
64       // Diálogo de confirmación
65       final confirmar = await showDialog<bool>(
66         context: context,
67         builder: (context) => AlertDialog(
68           title: const Text("¿Marcar como descanso semanal?"),
69           content: const Text(
70             "Confirma que el descanso actual es tu descanso semanal.\n\n"
71                 "• Si dura 45h o más → descanso normal ✅\n"
72                 "• Si dura entre 24h y 45h → descanso reducido ⚠️\n"
73                 "  (deberás compensar las horas restantes en 3 semanas)",
74           ),
75           actions: [
76             TextButton(
77               onPressed: () => Navigator.pop(context, false),
78               child: const Text("CANCELAR"),
79             ),
80             ElevatedButton(
81               onPressed: () => Navigator.pop(context, true),
82               child: const Text("CONFIRMAR"),
83             ),
84           ],
85         ),
86       );
87   
88       if (confirmar != true || !mounted) return;
89   
90       final res = await _api.marcarDescansoSemanal(
91         widget.idUltimoRegistro!,
92         widget.idUsuario,
93       );
94   
95       if (!mounted) return;
96   
97       if (res['status'] == 'success') {
98         ScaffoldMessenger.of(context).showSnackBar(
99           SnackBar(
100            content: Text(res['message']),
101            backgroundColor: Colors.green,
102          ),
103        );
104        _cargarEstado();
105      } else {
106        ScaffoldMessenger.of(context).showSnackBar(
107          SnackBar(
108            content: Text("Error: ${res['message']}"),
109            backgroundColor: Colors.red,
110          ),
111        );
112      }
113    }
114  
115    Color get _colorAlerta {
116      switch (_nivelAlerta) {
117        case 'critico': return Colors.red;
118        case 'aviso':   return Colors.orange;
119        default:        return Colors.green;
120      }
121    }
122  
123    IconData get _iconoAlerta {
124      switch (_nivelAlerta) {
125        case 'critico': return Icons.warning_rounded;
126        case 'aviso':   return Icons.info_rounded;
127        default:        return Icons.check_circle;
128      }
129    }
130  
131    @override
132    Widget build(BuildContext context) {
133      if (_cargando) return const LinearProgressIndicator();
134  
135      return Container(
136        width: double.infinity,
137        padding: const EdgeInsets.all(16),
138        decoration: BoxDecoration(
139          color: _colorAlerta.withOpacity(0.08),
140          borderRadius: BorderRadius.circular(12),
141          border: Border.all(color: _colorAlerta.withOpacity(0.4)),
142        ),
143        child: Column(
144          crossAxisAlignment: CrossAxisAlignment.start,
145          children: [
146            Row(
147              children: [
148                Icon(_iconoAlerta, color: _colorAlerta, size: 22),
149                const SizedBox(width: 8),
150                Text(
151                  "Descanso semanal",
152                  style: TextStyle(
153                    fontSize: 15,
154                    fontWeight: FontWeight.bold,
155                    color: _colorAlerta,
156                  ),
157                ),
158              ],
159            ),
160            const SizedBox(height: 12),
161  
162            // Barra de días trabajados
163            Row(
164              mainAxisAlignment: MainAxisAlignment.spaceBetween,
165              children: [
166                Text(
167                  "$_diasConsecutivos días trabajados",
168                  style: const TextStyle(fontSize: 13),
169                ),
170                Text(
171                  _diasHastaLimite > 0
172                      ? "Límite en $_diasHastaLimite días"
173                      : "¡Límite alcanzado!",
174                  style: TextStyle(
175                    fontSize: 13,
176                    fontWeight: FontWeight.bold,
177                    color: _colorAlerta,
178                  ),
179                ),
180              ],
181            ),
182            const SizedBox(height: 6),
183            ClipRRect(
184              borderRadius: BorderRadius.circular(4),
185              child: LinearProgressIndicator(
186                value: (_diasConsecutivos / 6).clamp(0.0, 1.0),
187                backgroundColor: Colors.grey.shade300,
188                valueColor: AlwaysStoppedAnimation<Color>(_colorAlerta),
189                minHeight: 10,
190              ),
191            ),
192            // Etiquetas de días
193            Row(
194              mainAxisAlignment: MainAxisAlignment.spaceBetween,
195              children: List.generate(7, (i) => Text(
196                i == 0 ? '' : '$i',
197                style: TextStyle(fontSize: 10, color: Colors.grey.shade500),
198              )),
199            ),
200  
201            // Compensaciones pendientes
202            if (_compensaciones.isNotEmpty) ...[
203              const Divider(height: 20),
204              Row(
205                children: [
206                  const Icon(Icons.access_time, size: 16, color: Colors.orange),
207                  const SizedBox(width: 6),
208                  Text(
209                    "Deuda pendiente: $_totalHorasDeuda horas",
210                    style: const TextStyle(
211                      fontSize: 13,
212                      fontWeight: FontWeight.bold,
213                      color: Colors.orange,
214                    ),
215                  ),
216                ],
217              ),
218              const SizedBox(height: 6),
219              ..._compensaciones.map((c) => Padding(
220                padding: const EdgeInsets.only(bottom: 4),
221                child: Text(
222                  "• ${c['horas_deuda']}h → antes del "
223                      "${c['fecha_limite'].toString().substring(0, 10)} "
224                      "(${c['dias_restantes']} días)",
225                  style: TextStyle(
226                    fontSize: 12,
227                    color: (c['dias_restantes'] as int) <= 3
228                        ? Colors.red
229                        : Colors.orange,
230                  ),
231                ),
232              )),
233            ],
234  
235            // Botón marcar descanso semanal
236            if (widget.idUltimoRegistro != null) ...[
237              const SizedBox(height: 12),
238              SizedBox(
239                width: double.infinity,
240                child: OutlinedButton.icon(
241                  icon: const Icon(Icons.hotel),
242                  label: const Text("Marcar como descanso semanal"),
243                  onPressed: _confirmarDescansoSemanal,
244                  style: OutlinedButton.styleFrom(
245                    foregroundColor: _colorAlerta,
246                    side: BorderSide(color: _colorAlerta),
247                  ),
248                ),
249              ),
250            ],
251          ],
252        ),
253      );
254    }
