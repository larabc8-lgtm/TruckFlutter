1    import 'package:flutter/material.dart';
2    import '../services/api_service.dart';
3    import '../services/session_service.dart';
4    import 'register_screen.dart';
5    import 'home_screen.dart';
6    
7    class LoginScreen extends StatefulWidget {
8      final VoidCallback onThemeToggle;
9      final VoidCallback? onLoginSuccess;
10   
11     const LoginScreen({
12       super.key,
13       required this.onThemeToggle,
14       this.onLoginSuccess,
15     });
16   
17     @override
18     State<LoginScreen> createState() => _LoginScreenState();
19   }
20   
21   class _LoginScreenState extends State<LoginScreen> {
22     final TextEditingController _emailController = TextEditingController();
23     final TextEditingController _passwordController = TextEditingController();
24     final ApiService _api = ApiService();
25   
26     bool _cargando = false;
27   
28     @override
29     void dispose() {
30       _emailController.dispose();
31       _passwordController.dispose();
32       super.dispose();
33     }
34   
35     void _intentarLogin() async {
36       setState(() => _cargando = true);
37   
38       var respuesta = await _api.login(
39         _emailController.text.trim(),
40         _passwordController.text.trim(),
41       );
42   
43       if (!mounted) return;
44       setState(() => _cargando = false);
45   
46       if (respuesta['status'] == 'success') {
47         await SessionService.guardarSesion(respuesta['data']);
48   
49         if (!mounted) return;
50         Navigator.pushReplacement(
51           context,
52           MaterialPageRoute(
53             builder: (context) => HomeScreen(
54               userData: respuesta['data'],
55               onLogout: () {
56                 SessionService.cerrarSesion();
57                 Navigator.pushAndRemoveUntil(
58                   context,
59                   MaterialPageRoute(
60                     builder: (context) => LoginScreen(
61                       onThemeToggle: widget.onThemeToggle,
62                       onLoginSuccess: widget.onLoginSuccess,
63                     ),
64                   ),
65                   (route) => false,
66                 );
67               },
68               onThemeToggle: widget.onThemeToggle,
69             ),
70           ),
71         );
72         widget.onLoginSuccess?.call();
73       } else {
74         ScaffoldMessenger.of(context).showSnackBar(
75           SnackBar(
76             content: Text("Error: ${respuesta['message']}"),
77             backgroundColor: Colors.red,
78           ),
79         );
80       }
81     }
82   
83     @override
84     Widget build(BuildContext context) {
85       return Scaffold(
86         appBar: AppBar(
87           title: const Text("TruckTime Login"),
88           actions: [
89             IconButton(
90               icon: Icon(
91                 Theme.of(context).brightness == Brightness.dark
92                     ? Icons.light_mode
93                     : Icons.dark_mode,
94               ),
95               onPressed: widget.onThemeToggle,
96               tooltip: "Cambiar tema",
97             ),
98           ],
99         ),
100        body: Padding(
101          padding: const EdgeInsets.all(20.0),
102          child: Column(
103            mainAxisAlignment: MainAxisAlignment.center,
104            children: [
105              const Icon(Icons.local_shipping, size: 80, color: Colors.blue),
106              const SizedBox(height: 10),
107              const Text(
108                "Bienvenido a TruckTime",
109                style: TextStyle(fontSize: 22, fontWeight: FontWeight.bold),
110              ),
111              const SizedBox(height: 30),
112              TextField(
113                controller: _emailController,
114                decoration: const InputDecoration(
115                  labelText: "Email",
116                  border: OutlineInputBorder(),
117                ),
118                keyboardType: TextInputType.emailAddress,
119              ),
120              const SizedBox(height: 20),
121              TextField(
122                controller: _passwordController,
123                decoration: const InputDecoration(
124                  labelText: "Contraseña",
125                  border: OutlineInputBorder(),
126                ),
127                obscureText: true,
128              ),
129              const SizedBox(height: 30),
130              _cargando
131                  ? const CircularProgressIndicator()
132                  : ElevatedButton(
133                      onPressed: _intentarLogin,
134                      style: ElevatedButton.styleFrom(
135                        minimumSize: const Size(double.infinity, 50),
136                      ),
137                      child: const Text("Iniciar Sesión"),
138                    ),
139              const SizedBox(height: 16),
140              TextButton(
141                onPressed: () => Navigator.push(
142                  context,
143                  MaterialPageRoute(
144                    builder: (context) =>
145                        RegisterScreen(onThemeToggle: widget.onThemeToggle),
146                  ),
147                ),
148                child: const Text("¿No tienes cuenta? Regístrate"),
149              ),
150            ],
151          ),
152        ),
153      );
154    }
155  }
